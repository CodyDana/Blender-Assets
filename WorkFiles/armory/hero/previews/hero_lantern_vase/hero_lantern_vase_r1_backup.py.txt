def _link(name, bm, mats):
    me = bpy.data.meshes.new(name)
    bm.to_mesh(me)
    bm.free()
    for m in mats:
        me.materials.append(bpy.data.materials[m])
    ob = bpy.data.objects.new(name, me)
    bpy.context.scene.collection.objects.link(ob)
    return ob


def _uv_project(bm, grain, tile=1.0):
    """Box-project UV0 in metres / tile, the grain axis along V wherever the face contains it."""
    uv = bm.loops.layers.uv.verify()
    ax = "xyz"
    for f in bm.faces:
        n = f.normal
        k = max(range(3), key=lambda i: abs(n[i]))
        plane = [i for i in range(3) if i != k]
        g = ax.index(grain)
        if g in plane:
            v_ax = g
            u_ax = [i for i in plane if i != g][0]
        else:
            u_ax, v_ax = plane
        for lp in f.loops:
            co = lp.vert.co
            lp[uv].uv = (co[u_ax] / tile, co[v_ax] / tile)


_WEAR = random.Random(5)


def _finish(piece, bm, grain, bevel, wear=0.6):
    """Chamfer every real arris (not the flat mitre / poke edges) by `bevel`; about `wear` of the chamfers take the
    worn-bronze EDGE, the rest stay FRAME. UV0 box projected along the grain; flat shaded."""
    bm.normal_update()
    if bevel:
        edges = [e for e in bm.edges if len(e.link_faces) == 2 and e.calc_face_angle(0) > 0.25]
        res = bmesh.ops.bevel(bm, geom=edges, offset=bevel, offset_type="OFFSET", segments=1, profile=0.5,
                              affect="EDGES", clamp_overlap=True)
        for f in res["faces"]:
            f.material_index = 1 if _WEAR.random() < wear else 0
    bm.normal_update()
    _uv_project(bm, grain, 1.0)
    H.from_object(piece, _link("tmp_frame", bm, [FRAME, EDGE]))


def _box_bm(x0, x1, y0, y1, z0, z1, apex=None):
    bm = bmesh.new()
    v = [bm.verts.new(c) for c in ((x0, y0, z0), (x1, y0, z0), (x1, y1, z0), (x0, y1, z0),
                                   (x0, y0, z1), (x1, y0, z1), (x1, y1, z1), (x0, y1, z1))]
    for q in ((0, 3, 2, 1), (0, 1, 5, 4), (2, 3, 7, 6), (3, 0, 4, 7), (1, 2, 6, 5)):
        bm.faces.new([v[i] for i in q])
    if apex is None:
        bm.faces.new([v[i] for i in (4, 5, 6, 7)])
    else:                       # a low hipped top: four faces rising to a centre apex
        ap = bm.verts.new(((x0 + x1) / 2, (y0 + y1) / 2, apex))
        for a, b in ((4, 5), (5, 6), (6, 7), (7, 4)):
            bm.faces.new((v[a], v[b], ap))
    return bm


def frame_box(piece, x0, x1, y0, y1, z0, z1, grain="z", bevel=0.002, apex=None, wear=0.6):
    _finish(piece, _box_bm(x0, x1, y0, y1, z0, z1, apex), grain, bevel, wear)


def frame_ring(piece, prof, bevel=0.002, wear=0.6):
    """A mitred square frame swept round the lantern axis: prof = its closed (half width, z) cross-section loop."""
    bm = bmesh.new()
    rings = [[bm.verts.new((sx * h, sy * h, z)) for sx, sy in ((-1, -1), (1, -1), (1, 1), (-1, 1))] for h, z in prof]
    n = len(prof)
    for k in range(n):
        a, b = rings[k], rings[(k + 1) % n]
        for i in range(4):
            j = (i + 1) % 4
            bm.faces.new((a[i], a[j], b[j], b[i]))
    bmesh.ops.recalc_face_normals(bm, faces=bm.faces)
    _finish(piece, bm, "x", bevel, wear)


def pane(piece, face, lo, hi, z0, z1, out0, out1):
    """A thin paper pane on one lantern face ('-y', '+y', '-x', '+x', or '+z' for the top pane: lo/hi are then the
    half extent and z0/z1 its thickness), the outer face mapped 0-1 (read left to right from outside), the other faces
    on one texel."""
    if face == "+z":
        x0, x1, y0, y1 = lo, hi, lo, hi
    else:
        a0, a1 = lo, hi
        if face == "-y":
            x0, x1, y0, y1 = a0, a1, -out1, -out0
        elif face == "+y":
            x0, x1, y0, y1 = a0, a1, out0, out1
        elif face == "-x":
            x0, x1, y0, y1 = -out1, -out0, a0, a1
        else:
            x0, x1, y0, y1 = out0, out1, a0, a1
    c = [(x0, y0, z0), (x1, y0, z0), (x1, y1, z0), (x0, y1, z0), (x0, y0, z1), (x1, y0, z1), (x1, y1, z1), (x0, y1, z1)]
    quads = {"-z": (0, 3, 2, 1), "+z": (4, 5, 6, 7), "-y": (0, 1, 5, 4), "+y": (2, 3, 7, 6), "-x": (3, 0, 4, 7),
             "+x": (1, 2, 6, 5)}
    faces, uvs = [], []
    for key, q in quads.items():
        faces.append(q)
        if key != face:
            uvs.append([(0.001, 0.001)] * 4)
            continue
        fu = []
        for i in q:
            x, y, z = c[i]
            if face == "-y":
                u, v = (x - x0) / (x1 - x0), (z - z0) / (z1 - z0)
            elif face == "+y":
                u, v = 1 - (x - x0) / (x1 - x0), (z - z0) / (z1 - z0)
            elif face == "-x":
                u, v = 1 - (y - y0) / (y1 - y0), (z - z0) / (z1 - z0)
            elif face == "+x":
                u, v = (y - y0) / (y1 - y0), (z - z0) / (z1 - z0)
            else:
                u, v = (x - x0) / (x1 - x0), (y - y0) / (y1 - y0)
            fu.append((u, v))
        uvs.append(fu)
    piece.mesh(c, faces, uvs, PAPER)

def _face_box(piece, face, a0, a1, o0, o1, z0, z1, grain, bevel=0.002, wear=0.6):
    if face == "-y":
        frame_box(piece, a0, a1, -o1, -o0, z0, z1, grain, bevel, wear=wear)
    elif face == "+y":
        frame_box(piece, a0, a1, o0, o1, z0, z1, grain, bevel, wear=wear)

        frame_box(piece, -o1, -o0, a0, a1, z0, z1, grain, bevel, wear=wear)

def lantern(G):
    ln = G["Piece"]("SM_AK_Lantern")
    for sx in (-1, 1):
        for sy in (-1, 1):
            cx, cy = sx * PX, sy * PX
            frame_box(ln, cx - PH, cx + PH, cy - PH, cy + PH, 0.047, 0.6395, "z", 0.0022, apex=0.6456)   # post, hip top
            frame_box(ln, cx - 0.0245, cx + 0.0245, cy - 0.0245, cy + 0.0245, 0.0, 0.049, "z", 0.0022)  # foot
    for face in ("-y", "+y", "-x", "+x"):
        along = "x" if face in ("-y", "+y") else "y"
        for z0, z1 in RAILS:
            _face_box(ln, face, -IN - 0.003, IN + 0.003, RAIL_OUT[0], RAIL_OUT[1], z0, z1, along)
        for z0, z1 in SPACERS:
            for s in (-1, 1):
                a0, a1 = (IN - 0.019, IN + 0.002) if s > 0 else (-IN - 0.002, -IN + 0.019)
                _face_box(ln, face, a0, a1, 0.1765, 0.1885, z0, z1, "z", 0.0012, wear=0.3)
    # cap board: 46 cm square frame, its inner edge stepped down 5 mm (the top view's groove line along each rail)
    frame_ring(ln, [(0.230, 0.590), (0.230, 0.609), (0.166, 0.609), (0.166, 0.604), (0.150, 0.604), (0.150, 0.590)],
               0.0016)
    # apron: a separate frame under the cap, its soffit sloping from the cap edge back to 5 mm behind the post faces
    frame_ring(ln, [(0.170, 0.564), (0.192, 0.564), (0.192, 0.5705), (0.226, 0.5905), (0.170, 0.5905)], 0.0018)
    frame_box(ln, -0.160, 0.160, -0.160, 0.160, 0.056, 0.066, "x", 0.0)                   # base board inside
    for face in ("-y", "+y", "-x", "+x"):
        pane(ln, face, -IN - 0.0015, IN + 0.0015, 0.1215, 0.5075, 0.166, 0.170)
    pane(ln, "+z", -0.172, 0.172, 0.5665, 0.5705, 0, 0)                                 # the pane under the cap
    ln.col(-0.23, 0.23, -0.23, 0.23, 0, 0.645)     # the scripted lantern's collision, unchanged
    return ln

def _frame(t):
    t = t.normalized()
    a = Vector((0, 0, 1)) if abs(t.z) < 0.9 else Vector((1, 0, 0))
    n = (a - t * a.dot(t)).normalized()
    return n, t.cross(n)


def tube(pts, radii, sides):
    """Tapered tube along pts (parallel-transport frames), closed by a pole at each end. -> verts, faces, uvs"""
    verts, faces, uvs, rings = [], [], [], []
    n = None
    for i, p in enumerate(pts):
        t = (pts[min(i + 1, len(pts) - 1)] - pts[max(i - 1, 0)]).normalized()
        if n is None:
            n, _b = _frame(t)
        n = (n - t * n.dot(t)).normalized()
        b = t.cross(n)
        ring = []
        for k in range(sides):
            a = 2 * math.pi * k / sides
            verts.append(tuple(p + radii[i] * (math.cos(a) * n + math.sin(a) * b)))
            ring.append(len(verts) - 1)
        rings.append(ring)
    L = [0.0]
    for i in range(1, len(pts)):
        L.append(L[-1] + (pts[i] - pts[i - 1]).length)
    for i in range(len(pts) - 1):
        a, c = rings[i], rings[i + 1]
        for k in range(sides):
            j = (k + 1) % sides
            faces.append((a[k], a[j], c[j], c[k]))
            uvs.append([(k / sides, L[i] * 8), ((k + 1) / sides, L[i] * 8), ((k + 1) / sides, L[i + 1] * 8),
                        (k / sides, L[i + 1] * 8)])
    t0 = (pts[0] - pts[1]).normalized()
    verts.append(tuple(pts[0] + t0 * radii[0] * 0.6))
    base = len(verts) - 1
    tip_d = (pts[-1] - pts[-2]).normalized()
    verts.append(tuple(pts[-1] + tip_d * max(radii[-1], 0.0015) * 1.5))
    tip = len(verts) - 1
    for k in range(sides):
        j = (k + 1) % sides
        faces.append((rings[0][j], rings[0][k], base))
        uvs.append([(0, 0), (0.1, 0), (0.05, 0.05)])
        faces.append((rings[-1][k], rings[-1][j], tip))
        uvs.append([(0, 0), (0.1, 0), (0.05, 0.05)])
    return verts, faces, uvs


def flower(c, nrm, R, petal_mat, rng, k=2, star=10, back_cap=True):
    """Five cupped round-ended petals (k rim points each) in two thin layers 0.8 mm apart (so the bloom reads from
    both sides with Unreal's one-sided flat materials, no coincident vertices), the back layer closed at the centre,
    and a raised ten-point yellow stamen star (the filaments) on top. Smooth shaded, planar UV.
    Tris: 2 x 5 x (2 + k) + 5 + 10."""
    nrm = nrm.normalized()
    u, v = _frame(nrm)
    th0 = rng.uniform(0, 2 * math.pi)
    cup = 0.40 * R
    verts, uvs2 = [], []

    def P(ang, r, h, off=0.0):
        verts.append(tuple(c + r * (math.cos(ang) * u + math.sin(ang) * v) + nrm * (h - off)))
        uvs2.append((0.5 + 0.5 * r / R * math.cos(ang - th0), 0.5 + 0.5 * r / R * math.sin(ang - th0)))
        return len(verts) - 1

    rims = {1: ((0, 1.0),), 2: ((-22, 0.95), (22, 0.95)), 3: ((-28, 0.9), (0, 1.0), (28, 0.9))}[k]
    mid_ang = {1: 30, 2: 33, 3: 35}[k]      # wide round petals, touching their neighbours near the rim
    layout = []      # per petal: angle, mid radii, rim radii (jittered once, shared by both layers)
    for i in range(5):
        a = th0 + math.radians(72 * i + rng.uniform(-4, 4))
        layout.append((a, [rng.uniform(0.5, 0.56) for _ in range(2)],
                       [rr * rng.uniform(0.94, 1.04) for _o, rr in rims]))
    faces, uvs, mats = [], [], []

    def add(f, m, flip=False):
        f = tuple(reversed(f)) if flip else tuple(f)
        faces.append(f)
        uvs.append([uvs2[j] for j in f])
        mats.append(m)

    for layer, off in ((0, 0.0), (1, 0.0008)):
        b = [P(th0 + math.radians(72 * i - 36), 0.16 * R, 0.012 * R + cup * 0.026, off) for i in range(5)]
        for i, (a, mids, rrs) in enumerate(layout):
            m0 = P(a - math.radians(mid_ang), mids[0] * R, cup * mids[0] ** 2, off)
            m1 = P(a + math.radians(mid_ang), mids[1] * R, cup * mids[1] ** 2, off)
            rv = [P(a + math.radians(o), rr * R, cup * rr ** 2 * 1.08, off) for (o, _r), rr in zip(rims, rrs)]
            b0, b1 = b[i], b[(i + 1) % 5]
            tris = [(b0, m0, m1), (b0, m1, b1)]
            chain = rv + [m1]
            for j in range(len(chain) - 1):
                tris.append((m0, chain[j], chain[j + 1]))
            for f in tris:
                add(f, petal_mat, flip=bool(layer))
        if layer and back_cap:     # close the back layer's centre
            cb = P(0, 0, 0.0, off)
            for i in range(5):
                add((cb, b[i], b[(i + 1) % 5]), petal_mat, flip=True)
    ctr = P(0, 0, 0.10 * R)
    if star == 10:      # filament tips and valleys
        ring = [P(th0 + math.radians(36 * j + 9), (0.44 if j % 2 == 0 else 0.18) * R,
                  (0.22 if j % 2 == 0 else 0.08) * R) for j in range(10)]
    else:               # small blooms: a raised five-point boss
        ring = [P(th0 + math.radians(72 * j + 36), 0.34 * R, 0.12 * R) for j in range(star)]
    for j in range(len(ring)):
        add((ctr, ring[j], ring[(j + 1) % len(ring)]), STAMEN)
    return verts, faces, uvs, mats


def bud(c, d, s):
    """A round plum bud: a smooth three-sided bipyramid, a little longer along d (6 tris)."""
    d = d.normalized()
    u, v = _frame(d)
    verts = [tuple(c + d * s * 1.25), tuple(c - d * s * 0.8)]
    for k in range(3):
        a = 2 * math.pi * k / 3
        verts.append(tuple(c + s * (math.cos(a) * u + math.sin(a) * v)))
    faces = []
    for k in range(3):
        j = (k + 1) % 3
        faces += [(0, 2 + k, 2 + j), (1, 2 + j, 2 + k)]
    uvs = [[(0.5, 1), (0, 0.5), (1, 0.5)] if f[0] == 0 else [(0.5, 0), (1, 0.5), (0, 0.5)] for f in faces]
    return verts, faces, uvs


def spray(cfg, hgt, fan_half, top, depth_half, fan_sign, front, seed):
    """Plum spray for one vase, fanned along local X (fan_sign -1: the sheet is seen from +Y, +1: from -Y).
    cfg: branches, mouth, r_main, kink, R (blossom radius range), k, red / white lists or dense stations, extra, buds.
    Returns a list of (verts, faces, uvs, mats, smooth) parts (fitted to the bbox afterwards by _fit)."""
    rng = random.Random(seed)
    branches = cfg["branches"]
    mx, my = cfg["mouth"]
    zb = hgt * 0.86                      # branch butts rest in the vase's hidden inner floor
    lmin = min(p[0] for b in branches for p in b[2]) - mx
    lmax = max(p[0] for b in branches for p in b[2]) - mx
    ytop = min(p[1] for b in branches for p in b[2])
    s_neg, s_pos = (fan_half - 0.012) / -lmin, (fan_half - 0.012) / lmax
    s_z = cfg.get("s_z") or 1.2 * (s_neg + s_pos) / 2
    if s_z == "fit":
        s_z = (top - hgt) / (my - ytop) * 0.97
    up = Vector((0, 0, 1))
    fanv = Vector((fan_sign, 0, 0))
    dep = Vector((0, 1, 0))

    def fanz(px, py):
        f = (px - mx) * (s_neg if px < mx else s_pos)
        return f, hgt + (my - py) * s_z

    def to3(f, z, d):
        return Vector((fan_sign * f, d, z))

    def planar(a):          # distance in the fan plane (depth ignored)
        return Vector((a.x, 0, a.z)).length

    skel, parts, tips = {}, [], []
    for name, parent, pts, rs, dtip in branches:
        fz = [fanz(*p) for p in pts]
        L = [0.0]
        for i in range(1, len(fz)):
            L.append(L[-1] + math.dist(fz[i - 1], fz[i]))
        if parent is None:
            ds = [dtip * depth_half * (l / L[-1]) ** 1.1 for l in L]
            p3 = [to3(fz[0][0] * 0.3, zb, ds[0] * 0.2)] + [to3(f, z, d) for (f, z), d in zip(fz, ds)]
            r0 = cfg["r_main"] * rs
        else:
            pp, pr, _pd = skel[parent]
            q = to3(*fz[0], 0)
            kk = min(range(len(pp)), key=lambda i: planar(pp[i] - q))
            d0 = pp[kk].y
            ds = [d0 + (dtip * depth_half - d0) * (l / L[-1]) for l in L]
            p3 = [pp[kk]] + [to3(f, z, d) for (f, z), d in zip(fz[1:], ds[1:])]
            r0 = pr[kk] * 0.72
        # subdivide into ~4.5 cm segments; plum wood zig-zags: alternate kinks at the nodes (mostly in the fan plane,
        # so the silhouette shows them) and a little random gnarl
        pts3 = []
        for i in range(len(p3) - 1):
            a, b = p3[i], p3[i + 1]
            n = max(1, int((b - a).length / cfg["seg"]))
            for j in range(n):
                pts3.append(a.lerp(b, j / n))
        pts3.append(p3[-1])
        side = rng.choice((-1, 1))
        for i in range(2, len(pts3) - 1):
            t = (pts3[i + 1] - pts3[i - 1]).normalized()
            perp = fanv + dep * rng.uniform(-0.5, 0.5)
            perp = (perp - t * perp.dot(t)).normalized()
            side = -side
            amp = cfg["kink"] * rng.uniform(0.4, 1.0)
            pts3[i] = pts3[i] + perp * side * amp + Vector([rng.uniform(-1, 1) for _ in range(3)]) * cfg["kink"] * 0.3
        total = sum((pts3[i + 1] - pts3[i]).length for i in range(len(pts3) - 1))
        radii, acc = [], 0.0
        for i, p in enumerate(pts3):
            if i:
                acc += (pts3[i] - pts3[i - 1]).length
            t = acc / total
            knob = rng.uniform(1.0, 1.22) if 1 < i < len(pts3) - 1 else 1.0     # knobbly nodes
            radii.append(max(0.0013, r0 * (1 - 0.9 * t ** 1.1) * knob))
        skel[name] = (pts3, radii, None)
        tips.append((pts3[-1], (pts3[-1] - pts3[-2]).normalized()))
    # short side twigs along the outer part of every branch, each ending in a bud (the node swells where they leave)
    twig_pts = []
    for name, (pts3, radii, _d) in list(skel.items()):
        total = sum((pts3[i + 1] - pts3[i]).length for i in range(len(pts3) - 1))
        acc, nxt, side = 0.0, total * rng.uniform(0.3, 0.4), rng.choice((-1, 1))
        for i in range(1, len(pts3)):
            acc += (pts3[i] - pts3[i - 1]).length
            if acc < nxt or acc > total - 0.03 or i == len(pts3) - 1:
                continue
            nxt = acc + rng.uniform(*cfg["twig_every"])
            side = -side
            t = (pts3[min(i + 1, len(pts3) - 1)] - pts3[i - 1]).normalized()
            perp = (fanv * 0.8 + dep * rng.uniform(-0.7, 0.7))
            perp = (perp - t * perp.dot(t)).normalized() * side
            ang = math.radians(rng.uniform(32, 52))
            d = (t * math.cos(ang) + perp * math.sin(ang)).normalized()
            ln_ = rng.uniform(*cfg["twig_len"])
            a = pts3[i]
            q = [a, a + d * ln_ * 0.5 + Vector([rng.uniform(-1, 1) for _ in range(3)]) * 0.003, a + d * ln_]
            r = [max(min(radii[i] * 0.5, 0.004), 0.0012), max(min(radii[i] * 0.3, 0.0025), 0.0011), 0.0011]
            radii[i] *= 1.15
            v, f, u = tube(q, r, 3)
            parts.append((v, f, u, BARK, True))
            tips.append((q[-1], d))
            twig_pts.append(q)
    for name, parent, pts, rs, dtip in branches:
        pts3, radii, _d = skel[name]
        sides = cfg["sides"][0] if parent is None and rs >= 0.8 else (cfg["sides"][1] if parent is None else 4)
        v, f, u = tube(pts3, radii, sides)
        parts.append((v, f, u, BARK, True))
    used = []

    def bloom(p, r, t, mat, R, bias):
        """A blossom on the branch at p (radius r, tangent t), opening outward from the branch toward bias."""
        o = Vector([rng.uniform(-1, 1) for _ in range(3)])
        o = (o - t * o.dot(t)).normalized()
        nrm = (o * 0.9 + bias + Vector([rng.uniform(-1, 1) for _ in range(3)]) * 0.25).normalized()
        c = p + nrm * (r + 0.0035)
        if any((c - q).length < R * cfg.get("space", 1.25) for q in used):
            return False
        used.append(c)
        parts.append((*flower(c, nrm, R, mat, rng, cfg["k"], cfg.get("star", 10), cfg.get("back_cap", True)), True))
        return True

    def nearest(target):
        best = None
        for pts3, radii, _d in skel.values():
            for i in range(1, len(pts3)):
                dd = planar(pts3[i] - target)
                if best is None or dd < best[0]:
                    t = (pts3[min(i + 1, len(pts3) - 1)] - pts3[i - 1]).normalized()
                    best = (dd, pts3[i], radii[i], t)
        return best[1:]

    R0, R1 = cfg["R"]
    if cfg.get("red"):
        # blossoms where the sheet draws them: on the nearest branch point, opening toward the viewer and outward
        for lst, mat in ((cfg["red"], PETAL), (cfg["white"], WHITE)):
            for (px, py) in lst:
                p, r, t = nearest(to3(*fanz(px, py), 0))
                bloom(p, r, t, mat, rng.uniform(R0, R1), front * 0.9)
        mains = [skel[b[0]] for b in branches if b[1] is None]
        extra, tries = cfg["extra"], 0
        while extra > 0 and tries < 400:
            tries += 1
            pts3, radii, _d = rng.choice(mains)
            i = rng.randrange(len(pts3) * 3 // 10, len(pts3) - 1)
            t = (pts3[i + 1] - pts3[i - 1]).normalized()
            if bloom(pts3[i], radii[i], t, PETAL if rng.random() < 0.85 else WHITE, rng.uniform(R0, R1) * 0.92,
                     front * 0.9):
                extra -= 1
    else:
        # dense: blossoms and buds all along the outer two thirds of every stem (back_wall's budding sprays)
        stations = []
        for name, (pts3, radii, _d) in skel.items():
            total = sum((pts3[i + 1] - pts3[i]).length for i in range(len(pts3) - 1))
            acc, nxt = 0.0, total * rng.uniform(0.22, 0.3)
            for i in range(1, len(pts3)):
                seg = (pts3[i] - pts3[i - 1]).length
                t = (pts3[i] - pts3[i - 1]).normalized()
                while nxt <= acc + seg:
                    stations.append((name, pts3[i - 1].lerp(pts3[i], (nxt - acc) / seg), radii[i], t))
                    nxt += rng.uniform(*cfg["dense"])
                acc += seg
        rng.shuffle(stations)          # the bloom cap thins every stem evenly
        nb = 0
        for name, p, r, t in stations:
            if rng.random() < cfg["bloom_frac"] and nb < cfg["max_blooms"]:
                white = (name in cfg["white_set"] and rng.random() < 0.55) or rng.random() < 0.05
                nb += bloom(p, r, t, WHITE if white else PETAL, rng.uniform(R0, R1), front * 0.9 + up * 0.2)
            else:
                o = Vector([rng.uniform(-1, 1) for _ in range(3)])
                o = (o - t * o.dot(t)).normalized()
                c = p + o * (r + 0.0045)
                if rng.random() < cfg["dense_buds"] and not any((c - q).length < 0.012 for q in used):
                    used.append(c)
                    parts.append((*bud(c, o * 0.5 + t, rng.uniform(0.0042, 0.0055)), BUD, True))
        print("HERO_BLOOMS", nb, "of", len(stations), "stations")
    # buds: every tip, plus a few along the outer twigs and branches
    for p, d in tips:
        parts.append((*bud(p + d * 0.004, d, rng.uniform(*cfg["bud_size"])), BUD, True))
    seats = [(pts3[i], radii[i], (pts3[i + 1] - pts3[i - 1]).normalized())
             for pts3, radii, _d in skel.values() for i in range(len(pts3) * 2 // 5, len(pts3) - 1)]
    seats += [(q[1], 0.0015, (q[2] - q[0]).normalized()) for q in twig_pts]
    for p, r, t in seats:
        if rng.random() < cfg["buds"]:
            o = Vector([rng.uniform(-1, 1) for _ in range(3)])
            o = (o - t * o.dot(t)).normalized()
            c = p + o * (r + 0.005)
            if any((c - q).length < 0.016 for q in used):
                continue
            used.append(c)
            parts.append((*bud(c, o * 0.6 + t, rng.uniform(*cfg["bud_size"]) * 0.9), BUD, True))
    return parts


def _fit(parts, hgt, lo, hi):
    """Scale the spray about the vase axis so it reaches the scripted bbox exactly along X and Y (each side on its own)
    and at the top (z above the mouth, cubic so the low fan keeps the sheet's shape)."""
    vs = [Vector(v) for p in parts for v in p[0]]
    mn = Vector([min(v[i] for v in vs) for i in range(3)])
    mxv = Vector([max(v[i] for v in vs) for i in range(3)])
    s = {}
    for i in range(2):
        s[(i, -1)], s[(i, 1)] = lo[i] / mn[i], hi[i] / mxv[i]
    h0 = mxv[2] - hgt
    sz = (hi[2] - hgt) / h0 - 1.0
    out = []
    for verts, faces, uvs, mats, sm in parts:
        nv = []
        for x, y, z in verts:
            x *= s[(0, -1)] if x < 0 else s[(0, 1)]
            y *= s[(1, -1)] if y < 0 else s[(1, 1)]
            if z > hgt:
                z = hgt + (z - hgt) * (1 + sz * ((z - hgt) / h0) ** 3)
            nv.append((x, y, z))
        out.append((nv, faces, uvs, mats, sm))
    return out, s, sz


def vase(G, name, hgt, r_vis, r_col, profile, sides, bbox_lo, bbox_hi, fan_sign, front, seed, cfg):
    """r_vis: the drawn vase's widest radius; r_col: the scripted vase's (its collision box, unchanged).
    bbox_lo / bbox_hi: the spray's envelope (the scripted cards' bbox, the sill vase pulled back on the wall side)."""
    v = G["Piece"](name)
    prof = [(r * r_vis, z * hgt) for r, z, _m in profile]
    verts, faces_v, uvs = G["lathe"](prof, sides)
    mats = [profile[k][2] for k in range(len(profile) - 1) for _i in range(sides)]
    assert len(mats) == len(faces_v)
    v.mesh(verts, faces_v, uvs, mats, smooth=True)
    parts = spray(cfg, hgt, bbox_hi[0], bbox_hi[2], bbox_hi[1] * 0.92, fan_sign, front, seed)
    parts, fs, fz = _fit(parts, hgt, bbox_lo, bbox_hi)
    tri = {}
    for verts_, faces, uvs_, m, sm in parts:
        for f, mm in zip(faces, m if isinstance(m, list) else [m] * len(faces)):
            tri[mm] = tri.get(mm, 0) + len(f) - 2
    print("HERO_TRIS", name, tri, "vase", sum(len(f) - 2 for f in faces_v))
    print("HERO_FIT", name, {str(k): round(x, 3) for k, x in fs.items()}, round(fz, 3))
    for verts_, faces, uvs_, m, sm in parts:
        v.mesh(verts_, faces, uvs_, m, smooth=sm)
    v.col(-r_col, r_col, -r_col, r_col, 0, hgt)          # the scripted vase's collision, unchanged
    return v

    rs_max = max(r for r, _ in G["vase_profile"](0.46))
